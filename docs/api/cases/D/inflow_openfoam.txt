    class       volVectorField;
    object      U;

{
    inlet
    {
        type            atmBoundaryLayerInletVelocity;
        flowDir         (1 0 0);
        zDir            (0 0 1);
        Uref            6.65;
        Zref            0.25;
        z0              uniform 8.59e-05;
        d               uniform 0;
        kappa           0.41;
        Cmu             0.09;
        value           $internalField;
