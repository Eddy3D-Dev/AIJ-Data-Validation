==== 0/U
/*--------------------------------*- C++ -*----------------------------------*\
  cases/G/inflow_openfoam.txt of the AIJ UWE API (release 3.3.1), written by scripts/build_api.py; the walk: https://aij.eddy3d.com/byos/
  RECOMMENDED inlet (this repository's choice, not the benchmark's): the measured approach-flow profile (approach_flow.csv, the benchmark's data) as a table; U is dimensionless, U/Ub (Ub = mean speed at 9 m, not published in m/s), z is m. flowDir is the case's wind direction by the convention along_x (cases/G.json inflow.flow_direction).
  OpenFOAM 12 (openfoam.org) syntax. ESI OpenFOAM v2xxx: the three atmBoundaryLayerInlet* conditions and nutkAtmRoughWallFunction take the same entries there; fixedProfile is openfoam.org only (use timeVaryingMappedFixedValue or codedFixedValue).
\*---------------------------------------------------------------------------*/
FoamFile
{
    format      ascii;
    class       volVectorField;
    object      U;
}
dimensions      [0 1 -1 0 0 0 0];
internalField   uniform (0 0 0);
boundaryField
{
    inlet
    {
        type            fixedProfile;
        direction       (0 0 1);
        origin          0;
        profile         table
        (
            (1.5 (0.69 0 0))
            (3 (0.77 0 0))
            (6 (0.9 0 0))
            (9 (1 0 0))
        );
    }
    outlet          { type inletOutlet; inletValue uniform (0 0 0); value $internalField; }
    ground          { type noSlip; }
    buildings       { type noSlip; }
    top             { type slip; }
    sides           { type slip; }
}
