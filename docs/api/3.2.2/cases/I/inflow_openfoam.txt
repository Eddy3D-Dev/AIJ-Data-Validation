    object      U;

{
    inlet
    {
        type            atmBoundaryLayerInletVelocity;
        flowDir         (1 0 0);
        zDir            (0 0 1);
        Uref            1.7;
        Zref            0.1;
        z0              uniform 4.55e-05;
        d               uniform 0;
        kappa           0.41;
        Cmu             0.09;
        value           $internalField;
    }
    outlet          { type inletOutlet; inletValue uniform (0 0 0); value $internalField; }
    ground          { type noSlip; }
